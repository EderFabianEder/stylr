<?php

namespace App\Http\Controllers\Api\V1\Auth;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class AuthController extends Controller
{
    public function user(Request $request): JsonResponse
    {
        $user = $request->user();
        $token = $user?->currentAccessToken();

        return response()->json([
            'success' => true,
            'data' => [
                'user' => [
                    'id' => $user->id,
                    'name' => $user->name,
                    'email' => $user->email,
                    'email_verified_at' => $user->email_verified_at,
                ],
                'current_token' => $token ? [
                    'name' => $token->name,
                    'last_used_at' => $token->last_used_at,
                    'expires_at' => $token->expires_at,
                ] : null,
            ],
        ]);
    }

    public function logout(Request $request): JsonResponse
    {
        $token = $request->user()?->currentAccessToken();

        if (!$token) {
            return response()->json([
                'success' => false,
                'message' => 'Kein aktiver Token gefunden.',
            ], 401);
        }

        $token->delete();

        return response()->json([
            'success' => true,
            'message' => 'Erfolgreich abgemeldet',
        ]);
    }

    public function logoutAll(Request $request): JsonResponse
    {
        $request->user()->tokens()->delete();

        return response()->json([
            'success' => true,
            'message' => 'Von allen GerÃ¤ten abgemeldet',
        ]);
    }

    public function refresh(Request $request): JsonResponse
    {
        $user = $request->user();
        $current = $user->currentAccessToken();

        if (!$current) {
            return response()->json([
                'success' => false,
                'message' => 'Kein aktiver Token gefunden.',
            ], 401);
        }

        $deviceName = $current->name;
        $abilities = $user->tokenAbilities();

        $current->delete();

        $expiresAt = now()->addDays(30);

        $newToken = $user->createToken($deviceName, $abilities, $expiresAt)->plainTextToken;

        return response()->json([
            'success' => true,
            'message' => 'Token erfolgreich erneuert',
            'data' => [
                'token' => $newToken,
                'token_type' => 'Bearer',
                'expires_at' => $expiresAt->toDateTimeString(),
            ],
        ]);
    }
}

<?php

namespace App\Http\Controllers\Api\V1\Auth;

use App\Http\Controllers\Controller;
use App\Models\User;
use Illuminate\Auth\Events\Verified;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class EmailVerificationController extends Controller
{
    public function send(Request $request): JsonResponse
    {
        $user = $request->user();

        if ($user->hasVerifiedEmail()) {
            return response()->json([
                'success' => false,
                'message' => 'Deine E-Mail-Adresse ist bereits verifiziert.',
            ], 400);
        }

        $user->sendEmailVerificationNotification();

        return response()->json([
            'success' => true,
            'message' => 'Verifizierungs-E-Mail wurde gesendet.',
        ]);
    }

    public function verify(Request $request, int $id, string $hash): JsonResponse
    {
        $user = User::findOrFail($id);

        $valid = hash_equals(sha1($user->getEmailForVerification()), $hash);

        if (!$valid) {
            return response()->json([
                'success' => false,
                'message' => 'UngÃ¼ltiger Verifizierungs-Link.',
            ], 403);
        }

        if ($user->hasVerifiedEmail()) {
            return response()->json([
                'success' => true,
                'message' => 'E-Mail-Adresse ist bereits verifiziert.',
            ]);
        }

        $user->markEmailAsVerified();

        return response()->json([
            'success' => true,
            'message' => 'E-Mail-Adresse erfolgreich verifiziert.',
        ]);
    }
}

<?php

namespace App\Http\Controllers\Api\V1\Auth;

use App\Http\Controllers\Controller;
use App\Models\User;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Laravel\Fortify\TwoFactorAuthenticationProvider;

class LoginController extends Controller
{
    private const MAX_LOGIN_ATTEMPTS = 5;
    private const LOCKOUT_MINUTES = 15;
    private const MAX_2FA_ATTEMPTS = 5;
    private const TWO_FACTOR_LOCKOUT_MINUTES = 10;
    private const TWO_FACTOR_CODE_DIGITS = 6;

    public function login(Request $request): JsonResponse
    {
        $request->validate([
            'email' => ['required', 'email'],
            'password' => ['required', 'string'],
            'device_name' => ['required', 'string', 'max:255'],
            'two_factor_code' => ['nullable', 'digits:' . self::TWO_FACTOR_CODE_DIGITS],
        ]);

        $user = User::where('email', $request->email)->first();

        if (! $user) {
            return response()->json([
                'success' => false,
                'message' => 'Die Anmeldedaten sind ungÃ¼ltig.',
            ], 422);
        }

        if ($user->locked_until && $user->locked_until->isFuture()) {
            return response()->json([
                'success' => false,
                'message' => 'Account temporÃ¤r gesperrt. Bitte spÃ¤ter erneut versuchen.',
                'data' => ['locked_until' => $user->locked_until->toDateTimeString()],
            ], 423);
        }

        if ($user->locked_until && $user->locked_until->isPast()) {
            $user->update(['failed_login_attempts' => 0, 'locked_until' => null]);
        }

        if (! Hash::check($request->password, $user->password)) {
            $tries = $user->failed_login_attempts + 1;
            $lockedUntil = $tries >= self::MAX_LOGIN_ATTEMPTS ? now()->addMinutes(self::LOCKOUT_MINUTES) : null;

            $user->update([
                'failed_login_attempts' => $tries,
                'locked_until' => $lockedUntil,
            ]);

            if ($lockedUntil) {
                return response()->json([
                    'success' => false,
                    'message' => 'Account temporÃ¤r gesperrt. Bitte spÃ¤ter erneut versuchen.',
                    'data' => ['locked_until' => $lockedUntil->toDateTimeString()],
                ], 423);
            }

            return response()->json([
                'success' => false,
                'message' => 'Die Anmeldedaten sind ungÃ¼ltig.',
                'data' => ['remaining_attempts' => self::MAX_LOGIN_ATTEMPTS - $tries],
            ], 422);
        }

        if ($user->two_factor_confirmed_at) {
            if ($user->two_factor_locked_until && $user->two_factor_locked_until->isFuture()) {
                return response()->json([
                    'success' => false,
                    'message' => '2FA temporÃ¤r gesperrt. Bitte spÃ¤ter erneut versuchen.',
                    'data' => [
                        'locked_until' => $user->two_factor_locked_until->toDateTimeString(),
                        'two_factor_locked' => true,
                    ],
                ], 423);
            }

            if ($user->two_factor_locked_until && $user->two_factor_locked_until->isPast()) {
                $user->update(['failed_two_factor_attempts' => 0, 'two_factor_locked_until' => null]);
            }

            if (! $request->filled('two_factor_code')) {
                return response()->json([
                    'success' => false,
                    'message' => '2FA erforderlich.',
                    'data' => ['two_factor_required' => true],
                ], 403);
            }

            if (! $user->two_factor_secret) {
                return response()->json([
                    'success' => false,
                    'message' => '2FA ist nicht korrekt konfiguriert.',
                ], 422);
            }

            $provider = app(TwoFactorAuthenticationProvider::class);
            $valid = $provider->verify(decrypt($user->two_factor_secret), $request->two_factor_code);

            if (! $valid) {
                $tries = $user->failed_two_factor_attempts + 1;
                $lockedUntil = $tries >= self::MAX_2FA_ATTEMPTS ? now()->addMinutes(self::TWO_FACTOR_LOCKOUT_MINUTES) : null;

                $user->update([
                    'failed_two_factor_attempts' => $tries,
                    'two_factor_locked_until' => $lockedUntil,
                ]);

                if ($lockedUntil) {
                    return response()->json([
                        'success' => false,
                        'message' => '2FA temporÃ¤r gesperrt. Bitte spÃ¤ter erneut versuchen.',
                        'data' => [
                            'locked_until' => $lockedUntil->toDateTimeString(),
                            'two_factor_locked' => true,
                        ],
                    ], 423);
                }

                return response()->json([
                    'success' => false,
                    'message' => '2FA-Code ist ungÃ¼ltig.',
                    'data' => [
                        'two_factor_required' => true,
                        'remaining_2fa_attempts' => self::MAX_2FA_ATTEMPTS - $tries,
                    ],
                ], 422);
            }
        }

        $user->update([
            'failed_login_attempts' => 0,
            'locked_until' => null,
            'failed_two_factor_attempts' => 0,
            'two_factor_locked_until' => null,
        ]);

        $expiresAt = now()->addDays(30);

        $token = $user->createToken(
            $request->device_name,
            $user->tokenAbilities(),
            $expiresAt
        )->plainTextToken;

        return response()->json([
            'success' => true,
            'message' => 'Login erfolgreich',
            'data' => [
                'user' => [
                    'id' => $user->id,
                    'name' => $user->name,
                    'email' => $user->email,
                    'email_verified_at' => $user->email_verified_at,
                ],
                'token' => $token,
                'token_type' => 'Bearer',
                'expires_at' => $expiresAt->toDateTimeString(),
                'abilities' => $user->tokenAbilities(),
            ],
        ]);
    }
}

<?php

namespace App\Http\Controllers\Api\V1\Auth;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Password;

class PasswordResetController extends Controller
{
    public function sendResetLink(Request $request): JsonResponse
    {
        $request->validate([
            'email' => ['required', 'email'],
        ]);

        $status = Password::sendResetLink($request->only('email'));

        return response()->json([
            'success' => true,
            'message' => 'Wenn ein Account existiert, wurde ein Reset-Link gesendet.',
        ]);
    }

    public function reset(Request $request): JsonResponse
    {
        $request->validate([
            'token' => ['required', 'string'],
            'email' => ['required', 'email'],
            'password' => ['required', 'string', 'confirmed', 'min:8'],
        ]);

        $status = Password::reset(
            $request->only('email', 'password', 'password_confirmation', 'token'),
            function ($user, $password) {
                $user->forceFill(['password' => $password])->save();
            }
        );

        if ($status !== Password::PASSWORD_RESET) {
            return response()->json([
                'success' => false,
                'message' => 'Reset fehlgeschlagen.',
            ], 422);
        }

        return response()->json([
            'success' => true,
            'message' => 'Dein Passwort wurde erfolgreich zurÃ¼ckgesetzt.',
        ]);
    }
}

<?php

namespace App\Http\Controllers\Api\V1\Auth;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Laravel\Fortify\Actions\ConfirmTwoFactorAuthentication;
use Laravel\Fortify\Actions\DisableTwoFactorAuthentication;
use Laravel\Fortify\Actions\EnableTwoFactorAuthentication;
use Laravel\Fortify\Actions\GenerateNewRecoveryCodes;

class TwoFactorController extends Controller
{
    private function requirePassword(Request $request): ?JsonResponse
    {
        $request->validate(['password' => ['required', 'string']]);

        if (!Hash::check($request->password, $request->user()->password)) {
            return response()->json([
                'success' => false,
                'message' => 'Passwort ist falsch.',
            ], 422);
        }

        return null;
    }

    public function enable(Request $request, EnableTwoFactorAuthentication $enable): JsonResponse
    {
        if ($res = $this->requirePassword($request)) return $res;

        $enable($request->user());

        return response()->json([
            'success' => true,
            'message' => '2FA aktiviert. Bitte QR-Code abrufen und danach bestÃ¤tigen.',
        ]);
    }

    public function qr(Request $request): JsonResponse
    {
        $user = $request->user();

        if (! $user->two_factor_secret) {
            return response()->json([
                'success' => false,
                'message' => '2FA ist nicht aktiviert.',
            ], 400);
        }

        return response()->json([
            'success' => true,
            'data' => [
                'svg' => $user->twoFactorQrCodeSvg(),
            ],
        ]);
    }

    public function confirm(Request $request, ConfirmTwoFactorAuthentication $confirm): JsonResponse
    {
        $request->validate([
            'code' => ['required', 'digits:6'],
        ]);

        $confirm($request->user(), $request->code);

        return response()->json([
            'success' => true,
            'message' => '2FA erfolgreich bestÃ¤tigt.',
        ]);
    }

    public function disable(Request $request, DisableTwoFactorAuthentication $disable): JsonResponse
    {
        if ($res = $this->requirePassword($request)) return $res;

        $disable($request->user());

        return response()->json([
            'success' => true,
            'message' => '2FA deaktiviert.',
        ]);
    }
}

<?php

namespace App\Http\Controllers\Api\V1\Moderation\Actions;

use App\Http\Controllers\Controller;
use App\Models\Ban;
use App\Models\Report;
use App\Models\User;
use App\Models\Warning;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class ModerationController extends Controller
{
    public function reports(): JsonResponse
    {
        $reports = Report::where('status', 'pending')
            ->with(['reporter:id,name', 'reportedUser:id,name', 'category:id,name'])
            ->latest()
            ->paginate(20);

        return response()->json([
            'success' => true,
            'data' => $reports,
        ]);
    }

    public function resolve(Request $request, Report $report): JsonResponse
    {
        if ($report->status !== 'pending') {
            return response()->json([
                'success' => false,
                'message' => 'Report bereits bearbeitet.',
            ], 400);
        }

        $user = $report->reportedUser;
        $admin = $request->user();

        Warning::create([
            'user_id' => $user->id,
            'category_id' => $report->category_id,
            'reason' => "Report bestÃ¤tigt: {$report->category->name}",
            'points' => $report->category->points,
            'type' => 'manual',
            'issued_by' => $admin->id,
            'is_active' => true,
        ]);

        $report->update([
            'status' => 'resolved',
            'handled_by' => $admin->id,
            'handled_at' => now(),
        ]);

        $points = (int) Warning::where('user_id', $user->id)
            ->where('is_active', true)
            ->sum('points');

        if ($points >= 6) {

            Ban::where('user_id', $user->id)
                ->where('is_active', true)
                ->update([
                    'is_active' => false,
                    'lifted_at' => now(),
                    'lifted_by' => $admin->id,
                    'lift_reason' => 'Ersetzt durch neue Sperre',
                ]);

            $until = now()->addHours(24);

            Ban::create([
                'user_id' => $user->id,
                'reason' => "Zu viele Punkte ({$points})",
                'type' => 'temporary',
                'starts_at' => now(),
                'until' => $until,
                'banned_by' => $admin->id,
                'is_active' => true,
            ]);

            $user->update([
                'is_banned' => true,
                'banned_until' => $until,
            ]);

            $user->tokens()->delete();
        }

        return response()->json([
            'success' => true,
            'message' => 'Report bestÃ¤tigt.',
        ]);
    }

    public function dismiss(Request $request, Report $report): JsonResponse
    {
        if ($report->status !== 'pending') {
            return response()->json([
                'success' => false,
                'message' => 'Report bereits bearbeitet.',
            ], 400);
        }

        $report->update([
            'status' => 'dismissed',
            'handled_by' => $request->user()->id,
            'handled_at' => now(),
        ]);

        return response()->json([
            'success' => true,
            'message' => 'Report abgelehnt.',
        ]);
    }

    public function ban(Request $request, User $user): JsonResponse
    {
        $request->validate([
            'reason' => ['required', 'string'],
            'hours' => ['nullable', 'integer', 'min:1'],
        ]);

        $admin = $request->user();
        $until = $request->hours ? now()->addHours($request->hours) : null;

        Ban::where('user_id', $user->id)
            ->where('is_active', true)
            ->update([
                'is_active' => false,
                'lifted_at' => now(),
                'lifted_by' => $admin->id,
                'lift_reason' => 'Ersetzt durch neue Sperre',
            ]);

        Ban::create([
            'user_id' => $user->id,
            'reason' => $request->reason,
            'type' => $until ? 'temporary' : 'permanent',
            'starts_at' => now(),
            'until' => $until,
            'banned_by' => $admin->id,
            'is_active' => true,
        ]);

        $user->update([
            'is_banned' => true,
            'banned_until' => $until,
        ]);

        $user->tokens()->delete();

        return response()->json([
            'success' => true,
            'message' => 'User gesperrt.',
        ]);
    }

    public function unban(Request $request, User $user): JsonResponse
    {
        $admin = $request->user();

        Ban::where('user_id', $user->id)
            ->where('is_active', true)
            ->update([
                'is_active' => false,
                'lifted_at' => now(),
                'lifted_by' => $admin->id,
                'lift_reason' => $request->reason ?? 'Manuell aufgehoben',
            ]);

        $user->update([
            'is_banned' => false,
            'banned_until' => null,
        ]);

        return response()->json([
            'success' => true,
            'message' => 'Sperre aufgehoben.',
        ]);
    }

<?php

namespace App\Http\Controllers\Api\V1\Moderation\Reports;

use App\Http\Controllers\Controller;
use App\Models\Ban;
use App\Models\Report;
use App\Models\ReportCategory;
use App\Models\User;
use App\Models\Warning;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class ReportController extends Controller
{
    public function categories(): JsonResponse
    {
        $categories = ReportCategory::where('is_active', true)
            ->orderBy('priority', 'desc')
            ->get(['slug', 'name', 'description']);

        return response()->json([
            'success' => true,
            'data' => $categories,
        ]);
    }

    public function store(Request $request, User $user): JsonResponse
    {
        $request->validate([
            'category' => ['required', 'string'],
            'reason' => ['nullable', 'string', 'max:500'],
            'content_type' => ['required', 'string', 'in:post,comment,message,profile'],
            'content_id' => ['required', 'integer'],
        ]);

        $reporter = $request->user();

        if ($reporter->id === $user->id) {
            return response()->json([
                'success' => false,
                'message' => 'Du kannst dich nicht selbst melden.',
            ], 400);
        }

        $category = ReportCategory::where('slug', $request->category)->first();

        if (!$category) {
            return response()->json([
                'success' => false,
                'message' => 'Kategorie nicht gefunden.',
            ], 404);
        }

        $alreadyReported = Report::where('reporter_id', $reporter->id)
            ->where('category_id', $category->id)
            ->where('content_type', $request->content_type)
            ->where('content_id', $request->content_id)
            ->exists();

        if ($alreadyReported) {
            return response()->json([
                'success' => false,
                'message' => 'Du hast diesen Inhalt bereits gemeldet.',
            ], 400);
        }

        $report = Report::create([
            'reporter_id' => $reporter->id,
            'reported_user_id' => $user->id,
            'category_id' => $category->id,
            'reason' => $request->reason,
            'content_type' => $request->content_type,
            'content_id' => $request->content_id,
            'status' => 'pending',
        ]);

        $reportCount = Report::where('reported_user_id', $user->id)
            ->where('category_id', $category->id)
            ->where('status', 'pending')
            ->count();

        if ($reportCount >= $category->threshold) {

            Warning::create([
                'user_id' => $user->id,
                'category_id' => $category->id,
                'reason' => "Automatisch: {$reportCount}x {$category->name}",
                'points' => $category->points,
                'type' => 'automatic',
                'is_active' => true,
            ]);

            Report::where('reported_user_id', $user->id)
                ->where('category_id', $category->id)
                ->where('status', 'pending')
                ->update([
                    'status' => 'resolved',
                    'handled_at' => now(),
                    'admin_note' => 'Auto-resolved',
                ]);

            $points = (int) Warning::where('user_id', $user->id)
                ->where('is_active', true)
                ->sum('points');

            if ($points >= 6) {
                $until = now()->addHours(24);

                Ban::create([
                    'user_id' => $user->id,
                    'reason' => "Automatisch: {$points} Punkte",
                    'type' => 'temporary',
                    'starts_at' => now(),
                    'until' => $until,
                    'is_active' => true,
                ]);

                $user->update([
                    'is_banned' => true,
                    'banned_until' => $until,
                ]);

                $user->tokens()->delete();
            }
        }

        return response()->json([
            'success' => true,
            'message' => 'Meldung eingereicht.',
            'data' => ['report_id' => $report->id],
        ], 201);
    }

    public function myWarnings(Request $request): JsonResponse
    {
        $warnings = Warning::where('user_id', $request->user()->id)
            ->with('category:id,name')
            ->latest()
            ->get();

        return response()->json([
            'success' => true,
            'data' => $warnings,
        ]);
    }

    public function myStatus(Request $request): JsonResponse
    {
        $user = $request->user();

        $points = (int) Warning::where('user_id', $user->id)
            ->where('is_active', true)
            ->sum('points');

        return response()->json([
            'success' => true,
            'data' => [
                'points' => $points,
                'max_points' => 6,
                'is_banned' => $user->is_banned,
                'banned_until' => $user->banned_until,
            ],
        ]);
    }
}

    private function canViewConnections(User $authUser, User $user): bool
    {
        if ($this->relationships->hasBlockRelation($authUser, $user)) {
            return false;
        }

        if ($user->isPublic()) {
            return true;
        }

        if ($authUser->id === $user->id) {
            return true;
        }

        return $this->relationships->isFollowing($authUser, $user);
    }

    public function follow(Request $request, User $user): JsonResponse
    {
        $authUser = $request->user();

        if ($authUser->id === $user->id) {
            return response()->json([
                'success' => false,
                'message' => 'Du kannst dir nicht selbst folgen.',
            ], 422);
        }

        if ($this->relationships->isFollowing($authUser, $user)) {
            return response()->json([
                'success' => false,
                'message' => 'Du folgst diesem User bereits.',
            ], 422);
        }

        if ($this->relationships->hasBlockRelation($authUser, $user)) {
            return response()->json([
                'success' => false,
                'message' => 'Aktion nicht mÃ¶glich.',
            ], 403);

                'followed_id' => $user->id,
            ]);

                ->where('requested_id', $user->id)
                ->delete();

            return response()->json([
                'success' => true,
                'message' => 'Du folgst diesem User jetzt.',
            ]);
        }

            return response()->json([
                'success' => false,
                'message' => 'Du hast bereits eine Anfrage gesendet.',
            ], 422);

            'success' => true,
            'message' => 'Anfrage gesendet. Warte auf BestÃ¤tigung.',

    public function unfollow(Request $request, User $user): JsonResponse
    {
        $authUser = $request->user();

        $deleted = Follow::where('follower_id', $authUser->id)
            ->where('followed_id', $user->id)
            ->delete();

        if ($deleted > 0) {
            return response()->json([
                'success' => true,
                'message' => 'Du folgst diesem User nicht mehr.',
            ]);
        }

        $deleted = FollowRequest::where('requester_id', $authUser->id)
            ->where('requested_id', $user->id)
            ->delete();

        if ($deleted > 0) {
            return response()->json([
                'success' => true,
                'message' => 'Anfrage zurÃ¼ckgezogen.',
            ]);
        }

        return response()->json([
            'success' => false,
            'message' => 'Keine Verbindung gefunden.',
        ], 404);
    }

        $followers = $user->followers()

    public function accept(Request $request, FollowRequest $followRequest): JsonResponse
    {
        if ($followRequest->requested_id !== $request->user()->id) {
            return response()->json([
                'success' => false,
                'message' => 'Das ist nicht deine Anfrage.',
            ], 403);
        }

        Follow::create([
            'follower_id' => $followRequest->requester_id,
            'followed_id' => $followRequest->requested_id,
        ]);

        $followRequest->delete();

        return response()->json([
            'success' => true,
            'message' => 'Anfrage angenommen. Der User folgt dir jetzt.',
        ]);
    }

    public function decline(Request $request, FollowRequest $followRequest): JsonResponse
    {
        if ($followRequest->requested_id !== $request->user()->id) {
            return response()->json([
                'success' => false,
                'message' => 'Das ist nicht deine Anfrage.',
            ], 403);
        }

        $followRequest->delete();

        return response()->json([
            'success' => true,
            'message' => 'Anfrage abgelehnt.',
        ]);
    }

    public function removeFollower(Request $request, User $user): JsonResponse
    {
        $deleted = Follow::where('follower_id', $user->id)
            ->where('followed_id', $request->user()->id)
            ->delete();

        if ($deleted === 0) {
            return response()->json([
                'success' => false,
                'message' => 'User folgt dir nicht.',
            ], 404);
        }

        return response()->json([
            'success' => true,
            'message' => 'Follower entfernt.',
        ]);
    }

    public function show(Request $request, User $user): JsonResponse
    {
        $authUser = $request->user();

        $data = [
            'id' => $user->id,
            'name' => $user->name,
            'account_type' => $user->account_type,
            'followers_count' => $user->followers()->count(),
            'following_count' => $user->following()->count(),
            'is_following' => $this->relationships->isFollowing($authUser, $user),
            'has_pending_request' => $this->relationships->hasSentRequestTo($authUser, $user),
        ];

        if ($this->canViewConnections($authUser, $user)) {
            $data['is_followed_by'] = $this->relationships->isFollowedBy($authUser, $user);
            $data['is_mutual'] = $this->relationships->isMutual($authUser, $user);
        }

        return response()->json([
            'success' => true,
            'data' => $data,
        ]);
    }
